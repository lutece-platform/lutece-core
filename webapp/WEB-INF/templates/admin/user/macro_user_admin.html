<#-- # Shared pieces of the back office screens of the administrators (ManageUsers.jsp and the screens reached from it): # userBreadcrumb the breadcrumb: the list of the users, then the current screen # param title the page title (a capture or an already localised text) # param iconName the Tabler icon of the avatar # param breadcrumb the breadcrumb (see userBreadcrumb) # param userId the id of the user a screen is about: its avatar replaces the one of the feature when the avatar plugin is there # The nested content is the tools of the header (buttons, menus). --> <#macro userPageHeader title iconName='user-cog' breadcrumb='' userId=''> <#local pageTitle><#if userId?is_number && ( adminAvatar!false )?is_boolean && ( adminAvatar!false )><@img url='servlet/plugins/adminavatar/avatar?id_user=${userId?c}' class='avatar avatar-sm me-2' alt='' /><#else><@icon style=iconName />${title} <@pageHeader title=pageTitle breadcrumb=breadcrumb> <#nested> <#-- # param active the current tab: user, password, rights, roles or workgroups # param userId the id of the user # param password false when the authentication module manages the passwords (no password tab) --> <#macro userTabs active userId password=true> <@tabs> <@tabList> <@tabLink active=( active == 'user' ) href='jsp/admin/user/ModifyUser.jsp?id_user=${userId?c}' title='#i18n{portal.users.actions.labelUser}' /> <#if password><@tabLink active=( active == 'password' ) href='jsp/admin/user/ModifyUserPassword.jsp?id_user=${userId?c}' title='#i18n{portal.users.actions.labelPassword}' /> <@tabLink active=( active == 'rights' ) href='jsp/admin/user/ModifyUserRights.jsp?id_user=${userId?c}' title='#i18n{portal.users.actions.labelRights}' /> <@tabLink active=( active == 'roles' ) href='jsp/admin/user/ModifyUserRoles.jsp?id_user=${userId?c}' title='#i18n{portal.users.actions.labelRoles}' /> <@tabLink active=( active == 'workgroups' ) href='jsp/admin/user/ModifyUserWorkgroups.jsp?id_user=${userId?c}' title='#i18n{portal.users.actions.labelWorkgroups}' /> <@tabContent> <#nested>
